
import {
  DataTypes,
  Model,
  Optional,
} from "sequelize";
import { sequelize } from "../config/database.js";

interface NutricionistaAttributes {
  id: string;
  nome: string;
  email: string;
  senha: string;
  createdAt?: Date;
  updatedAt?: Date;
}

type NutricionistaCreationAttributes = Optional<
  NutricionistaAttributes,
  "id" | "createdAt" | "updatedAt"
>;

export class Nutricionista
  extends Model<
    NutricionistaAttributes,
    NutricionistaCreationAttributes
  >
  implements NutricionistaAttributes
{
  declare id: string;
  declare nome: string;
  declare email: string;
  declare senha: string;
  declare createdAt: Date;
  declare updatedAt: Date;
}

Nutricionista.init(
  {
    id: {
      type: DataTypes.UUID,
      defaultValue: DataTypes.UUIDV4,
      primaryKey: true,
    },
    nome: {
      type: DataTypes.STRING(150),
      allowNull: false,
    },
    email: {
      type: DataTypes.STRING(255),
      allowNull: false,
      unique: true,
      validate: {
        isEmail: true,
      },
    },
    senha: {
      type: DataTypes.STRING(255),
      allowNull: false,
    },
  },
  {
    sequelize,
    tableName: "nutricionistas",
    timestamps: true,
  }
);
